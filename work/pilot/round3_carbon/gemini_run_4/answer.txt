import java.util.ArrayList;
import java.util.Collections;
import java.util.HashSet;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.TreeMap;

public final class WalRecovery {
    private WalRecovery() {}

    private static final byte[] MAGIC = {0x57, 0x41, 0x4C, 0x31};
    private static final int HEADER_SIZE = 4;
    private static final int MAX_PAYLOAD = 1024;
    private static final int MIN_KEY_LEN = 1;
    private static final int MAX_KEY_LEN = 32;
    private static final int KEY_BYTE_MIN = 0x21;
    private static final int KEY_BYTE_MAX = 0x7E;
    private static final int VALUE_BYTE_MIN = 0x20;
    private static final int VALUE_BYTE_MAX = 0x7E;
    private static final int MAX_OPEN_TXNS = 4;
    private static final int MAX_WRITES = 8;
    private static final int NAME_BYTE_MIN = 0x21;
    private static final int NAME_BYTE_MAX = 0x7E;

    private static final int TYPE_BEGIN = 1;
    private static final int TYPE_PUT = 2;
    private static final int TYPE_DELETE = 3;
    private static final int TYPE_COMMIT = 4;
    private static final int TYPE_ABORT = 5;
    private static final int TYPE_SAVEPOINT = 6;
    private static final int TYPE_ROLLBACK = 7;
    private static final int TYPE_CHECKPOINT = 8;

    private record WriteOp(boolean isPut, String key, String value) {}

    private static final class Savepoint {
        int position;
        int recordOffset;

        Savepoint(int position, int recordOffset) {
            this.position = position;
            this.recordOffset = recordOffset;
        }
    }

    private static final class OpenTxn {
        final List<WriteOp> writes = new ArrayList<>();
        final Map<Integer, Savepoint> savepoints = new LinkedHashMap<>();
    }

    /** Recovers the store from a log. Must not modify {@code log}. */
    public static Result recover(byte[] log) throws WalException {
        // R1
        if (log == null) {
            throw new NullPointerException();
        }

        final int L = log.length;

        // R2
        if (L == 0) {
            return emptyResult(0, false);
        }

        // R3
        if (L < HEADER_SIZE) {
            for (int i = 0; i < L; i++) {
                if (log[i] != MAGIC[i]) {
                    throw new WalException(Code.BAD_MAGIC, 0);
                }
            }
            return emptyResult(0, true);
        }

        // R4
        for (int i = 0; i < HEADER_SIZE; i++) {
            if (log[i] != MAGIC[i]) {
                throw new WalException(Code.BAD_MAGIC, 0);
            }
        }

        // Phase 1: frames
        List<Integer> intactOffsets = new ArrayList<>();
        int p = HEADER_SIZE;
        int validLength;
        boolean tornTail;

        while (true) {
            // R5
            if (p == L) {
                validLength = L;
                tornTail = false;
                break;
            }
            // R6
            if (L - p <= 2) {
                validLength = p;
                tornTail = true;
                break;
            }

            int type = u8(log, p);
            // R7
            if (type < TYPE_BEGIN || type > TYPE_CHECKPOINT) {
                throw new WalException(Code.UNKNOWN_TYPE, p);
            }

            int N = u16(log, p + 1);
            // R8
            if (N > MAX_PAYLOAD) {
                throw new WalException(Code.RECORD_TOO_LARGE, p);
            }

            int nextP = p + 5 + N;
            // R9
            if (nextP > L) {
                validLength = p;
                tornTail = true;
                break;
            }

            // R10: Fletcher-16 checksum over offsets p .. p + 2 + N
            int s1 = 0;
            int s2 = 0;
            int end = p + 3 + N;
            for (int i = p; i < end; i++) {
                s1 = (s1 + (log[i] & 0xFF)) % 255;
                s2 = (s2 + s1) % 255;
            }
            int storedS2 = u8(log, end);
            int storedS1 = u8(log, end + 1);
            if (storedS2 != s2 || storedS1 != s1) {
                if (nextP == L) {
                    validLength = p;
                    tornTail = true;
                    break;
                } else {
                    throw new WalException(Code.CHECKSUM_MISMATCH, p);
                }
            }

            // R11
            intactOffsets.add(p);
            p = nextP;
        }

        // Phase 2: replay
        Map<String, String> state = new LinkedHashMap<>();
        List<Long> committed = new ArrayList<>();
        List<Long> aborted = new ArrayList<>();
        Set<Long> seenTxns = new HashSet<>();
        TreeMap<Long, OpenTxn> openTxns = new TreeMap<>();

        for (int recOffset : intactOffsets) {
            int type = u8(log, recOffset);
            int N = u16(log, recOffset + 1);
            int pb = recOffset + 3; // payload base

            // R13: Payload layout
            int k = 0;
            int v = 0;
            int c = 0;
            switch (type) {
                case TYPE_BEGIN, TYPE_COMMIT, TYPE_ABORT -> {
                    if (N != 4) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                }
                case TYPE_PUT -> {
                    if (N < 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                    k = u8(log, pb + 4);
                    if (N < 7 + k) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                    v = u16(log, pb + 5 + k);
                    if (N != 7 + k + v) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                }
                case TYPE_DELETE -> {
                    if (N < 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                    k = u8(log, pb + 4);
                    if (N != 5 + k) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                }
                case TYPE_SAVEPOINT, TYPE_ROLLBACK -> {
                    if (N != 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                }
                case TYPE_CHECKPOINT -> {
                    if (N < 1) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                    c = u8(log, pb);
                    if (N != 1 + 4 * c) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, recOffset);
                    }
                }
            }

            // R14: Txn ids
            long txn = 0;
            long[] checkpointIds = null;
            if (type == TYPE_CHECKPOINT) {
                checkpointIds = new long[c];
                for (int i = 0; i < c; i++) {
                    long listedId = u32(log, pb + 1 + 4 * i);
                    if (listedId == 0) {
                        throw new WalException(Code.BAD_TXN_ID, recOffset);
                    }
                    checkpointIds[i] = listedId;
                }
            } else {
                txn = u32(log, pb);
                if (txn == 0) {
                    throw new WalException(Code.BAD_TXN_ID, recOffset);
                }
            }

            // R15: Key (PUT, DELETE)
            String key = null;
            if (type == TYPE_PUT || type == TYPE_DELETE) {
                if (k < MIN_KEY_LEN || k > MAX_KEY_LEN) {
                    throw new WalException(Code.BAD_KEY, recOffset);
                }
                char[] keyChars = new char[k];
                for (int i = 0; i < k; i++) {
                    int b = u8(log, pb + 5 + i);
                    if (b < KEY_BYTE_MIN || b > KEY_BYTE_MAX) {
                        throw new WalException(Code.BAD_KEY, recOffset);
                    }
                    keyChars[i] = (char) b;
                }
                key = new String(keyChars);
            }

            // R16: Value (PUT)
            String value = null;
            if (type == TYPE_PUT) {
                char[] valChars = new char[v];
                int valStart = pb + 7 + k;
                for (int i = 0; i < v; i++) {
                    int b = u8(log, valStart + i);
                    if (b < VALUE_BYTE_MIN || b > VALUE_BYTE_MAX) {
                        throw new WalException(Code.BAD_VALUE, recOffset);
                    }
                    valChars[i] = (char) b;
                }
                value = new String(valChars);
            }

            // R16a: Name (SAVEPOINT, ROLLBACK)
            int spName = 0;
            if (type == TYPE_SAVEPOINT || type == TYPE_ROLLBACK) {
                spName = u8(log, pb + 4);
                if (spName < NAME_BYTE_MIN || spName > NAME_BYTE_MAX) {
                    throw new WalException(Code.BAD_NAME, recOffset);
                }
            }

            // R17 & R18: BEGIN checks
            if (type == TYPE_BEGIN) {
                if (seenTxns.contains(txn)) {
                    throw new WalException(Code.DUPLICATE_TXN, recOffset);
                }
                if (openTxns.size() >= MAX_OPEN_TXNS) {
                    throw new WalException(Code.TOO_MANY_OPEN, recOffset);
                }
            }

            // R19: Open check (PUT, DELETE, COMMIT, ABORT, SAVEPOINT, ROLLBACK)
            OpenTxn openTxn = null;
            if (type >= TYPE_PUT && type <= TYPE_ROLLBACK) {
                openTxn = openTxns.get(txn);
                if (openTxn == null) {
                    throw new WalException(Code.TXN_NOT_OPEN, recOffset);
                }
            }

            // R19a: Write limit (PUT, DELETE)
            if (type == TYPE_PUT || type == TYPE_DELETE) {
                if (openTxn.writes.size() >= MAX_WRITES) {
                    throw new WalException(Code.TXN_TOO_LARGE, recOffset);
                }
            }

            // R19b: Savepoint check (ROLLBACK)
            Savepoint targetSp = null;
            if (type == TYPE_ROLLBACK) {
                targetSp = openTxn.savepoints.get(spName);
                if (targetSp == null) {
                    throw new WalException(Code.NO_SAVEPOINT, recOffset);
                }
            }

            // R19c: Checkpoint check (CHECKPOINT)
            if (type == TYPE_CHECKPOINT) {
                if (c != openTxns.size()) {
                    throw new WalException(Code.CHECKPOINT_MISMATCH, recOffset);
                }
                int idx = 0;
                for (Long openId : openTxns.keySet()) {
                    if (checkpointIds[idx++] != openId) {
                        throw new WalException(Code.CHECKPOINT_MISMATCH, recOffset);
                    }
                }
            }

            // Actions: R20 to R23c
            switch (type) {
                case TYPE_BEGIN -> {
                    seenTxns.add(txn);
                    openTxns.put(txn, new OpenTxn());
                }
                case TYPE_PUT -> openTxn.writes.add(new WriteOp(true, key, value));
                case TYPE_DELETE -> openTxn.writes.add(new WriteOp(false, key, null));
                case TYPE_COMMIT -> {
                    for (WriteOp op : openTxn.writes) {
                        if (op.isPut()) {
                            state.put(op.key(), op.value());
                        } else {
                            state.remove(op.key());
                        }
                    }
                    committed.add(txn);
                    openTxns.remove(txn);
                }
                case TYPE_ABORT -> {
                    aborted.add(txn);
                    openTxns.remove(txn);
                }
                case TYPE_SAVEPOINT -> {
                    int pos = openTxn.writes.size();
                    Savepoint existing = openTxn.savepoints.get(spName);
                    if (existing == null) {
                        openTxn.savepoints.put(spName, new Savepoint(pos, recOffset));
                    } else {
                        existing.position = pos;
                        existing.recordOffset = recOffset;
                    }
                }
                case TYPE_ROLLBACK -> {
                    int n = targetSp.position;
                    openTxn.writes.subList(n, openTxn.writes.size()).clear();
                    int sOffset = targetSp.recordOffset;
                    openTxn.savepoints.values().removeIf(sp -> sp.recordOffset > sOffset);
                }
                case TYPE_CHECKPOINT -> {
                    // R23c: nothing changes
                }
            }
        }

        List<Long> incomplete = List.copyOf(openTxns.keySet());
        return new Result(
                Collections.unmodifiableMap(state),
                List.copyOf(committed),
                List.copyOf(aborted),
                incomplete,
                validLength,
                tornTail
        );
    }

    private static Result emptyResult(int validLength, boolean tornTail) {
        return new Result(Map.of(), List.of(), List.of(), List.of(), validLength, tornTail);
    }

    private static int u8(byte[] log, int offset) {
        return log[offset] & 0xFF;
    }

    private static int u16(byte[] log, int offset) {
        return ((log[offset] & 0xFF) << 8) | (log[offset + 1] & 0xFF);
    }

    private static long u32(byte[] log, int offset) {
        return ((long) (log[offset] & 0xFF) << 24)
                | ((long) (log[offset + 1] & 0xFF) << 16)
                | ((long) (log[offset + 2] & 0xFF) << 8)
                | ((long) (log[offset + 3] & 0xFF));
    }

    public record Result(Map<String, String> state,
                         List<Long> committed,
                         List<Long> aborted,
                         List<Long> incomplete,
                         int validLength,
                         boolean tornTail) {}

    public enum Code {
        BAD_MAGIC, UNKNOWN_TYPE, RECORD_TOO_LARGE, CHECKSUM_MISMATCH,
        MALFORMED_PAYLOAD, BAD_TXN_ID, BAD_KEY, BAD_VALUE,
        DUPLICATE_TXN, TOO_MANY_OPEN, TXN_NOT_OPEN,
        TXN_TOO_LARGE, BAD_NAME, NO_SAVEPOINT,
        CHECKPOINT_MISMATCH
    }

    public static final class WalException extends Exception {
        private final Code code;
        private final int offset;
        public WalException(Code code, int offset) {
            super(code + "@" + offset);
            this.code = code;
            this.offset = offset;
        }
        public Code code() { return code; }
        public int offset() { return offset; }
    }
}
