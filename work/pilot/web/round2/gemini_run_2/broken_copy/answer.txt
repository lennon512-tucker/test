import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Iterator;
import java.util.List;
import java.util.Map;
import java.util.Set;

public final class WalRecovery {
    private WalRecovery() {}

    private static final byte[] MAGIC = {0x57, 0x41, 0x4C, 0x31};
    private static final int HEADER_SIZE = 4;
    private static final int MAX_PAYLOAD = 1024;
    private static final int MIN_KEY_LEN = 1;
    private static final int MAX_KEY_LEN = 32;
    private static final int KEY_BYTE_MIN = 0x21;
    private static final int KEY_BYTE_MAX = 0x7E;
    private static final int VALUE_BYTE_MIN = 0x20;
    private static final int VALUE_BYTE_MAX = 0x7E;
    private static final int MAX_OPEN_TXNS = 4;
    private static final int MAX_WRITES = 8;
    private static final int NAME_BYTE_MIN = 0x21;
    private static final int NAME_BYTE_MAX = 0x7E;

    /** Recovers the store from a log. Must not modify {@code log}. */
    public static Result recover(byte[] log) throws WalException {
        // R1
        if (log == null) {
            throw new NullPointerException("log must not be null");
        }
        int L = log.length;

        // R2
        if (L == 0) {
            return new Result(Map.of(), List.of(), List.of(), List.of(), 0, false);
        }

        // R3
        if (L < HEADER_SIZE) {
            for (int i = 0; i < L; i++) {
                if (log[i] != MAGIC[i]) {
                    throw new WalException(Code.BAD_MAGIC, 0);
                }
            }
            return new Result(Map.of(), List.of(), List.of(), List.of(), 0, true);
        }

        // R4
        for (int i = 0; i < HEADER_SIZE; i++) {
            if (log[i] != MAGIC[i]) {
                throw new WalException(Code.BAD_MAGIC, 0);
            }
        }

        // Phase 1: frames
        List<RecordFrame> records = new ArrayList<>();
        int p = HEADER_SIZE;
        int validLength;
        boolean tornTail;

        while (true) {
            // R5
            if (p == L) {
                validLength = L;
                tornTail = false;
                break;
            }
            // R6
            if (L - p <= 2) {
                validLength = p;
                tornTail = true;
                break;
            }
            // R7
            int type = u8(log, p);
            if (type < 1 || type > 7) {
                throw new WalException(Code.UNKNOWN_TYPE, p);
            }
            // R8
            int n = u16(log, p + 1);
            if (n > MAX_PAYLOAD) {
                throw new WalException(Code.RECORD_TOO_LARGE, p);
            }
            // R9
            int end = p + 5 + n;
            if (end > L) {
                validLength = p;
                tornTail = true;
                break;
            }
            // R10
            int s1 = 0;
            int s2 = 0;
            int checksumStart = p + 3 + n;
            for (int i = p; i < checksumStart; i++) {
                int b = u8(log, i);
                s1 = (s1 + b) % 255;
                s2 = (s2 + s1) % 255;
            }
            int storedS2 = u8(log, checksumStart);
            int storedS1 = u8(log, checksumStart + 1);
            if (storedS2 != s2 || storedS1 != s1) {
                if (end == L) {
                    validLength = p;
                    tornTail = true;
                    break;
                } else {
                    throw new WalException(Code.CHECKSUM_MISMATCH, p);
                }
            }
            // R11
            records.add(new RecordFrame(p, type, n));
            p = end;
        }

        // Phase 2: replay
        Map<String, String> state = new HashMap<>();
        List<Long> committed = new ArrayList<>();
        List<Long> aborted = new ArrayList<>();
        Set<Long> seenTxns = new HashSet<>();
        Map<Long, TxnState> openTxns = new HashMap<>();

        for (RecordFrame rec : records) {
            int offset = rec.offset;
            int type = rec.type;
            int n = rec.length;
            int payloadOffset = offset + 3;

            // R13: Payload layout
            switch (type) {
                case 1, 4, 5 -> { // BEGIN, COMMIT, ABORT
                    if (n != 4) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                }
                case 2 -> { // PUT
                    if (n < 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                    int k = u8(log, payloadOffset + 4);
                    if (n < 7 + k) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                    int v = u16(log, payloadOffset + 5 + k);
                    if (n != 7 + k + v) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                }
                case 3 -> { // DELETE
                    if (n < 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                    int k = u8(log, payloadOffset + 4);
                    if (n != 5 + k) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                }
                case 6, 7 -> { // SAVEPOINT, ROLLBACK
                    if (n != 5) {
                        throw new WalException(Code.MALFORMED_PAYLOAD, offset);
                    }
                }
                default -> throw new AssertionError("unreachable");
            }

            // R14: Txn id
            long txn = u32(log, payloadOffset);
            if (txn == 0L) {
                throw new WalException(Code.BAD_TXN_ID, offset);
            }

            // R15: Key (PUT, DELETE)
            String key = null;
            if (type == 2 || type == 3) {
                int k = u8(log, payloadOffset + 4);
                if (k < MIN_KEY_LEN || k > MAX_KEY_LEN) {
                    throw new WalException(Code.BAD_KEY, offset);
                }
                char[] keyChars = new char[k];
                for (int i = 0; i < k; i++) {
                    int b = u8(log, payloadOffset + 5 + i);
                    if (b < KEY_BYTE_MIN || b > KEY_BYTE_MAX) {
                        throw new WalException(Code.BAD_KEY, offset);
                    }
                    keyChars[i] = (char) b;
                }
                key = new String(keyChars);
            }

            // R16: Value (PUT)
            String value = null;
            if (type == 2) {
                int k = u8(log, payloadOffset + 4);
                int v = u16(log, payloadOffset + 5 + k);
                char[] valChars = new char[v];
                int valStart = payloadOffset + 7 + k;
                for (int i = 0; i < v; i++) {
                    int b = u8(log, valStart + i);
                    if (b < VALUE_BYTE_MIN || b > VALUE_BYTE_MAX) {
                        throw new WalException(Code.BAD_VALUE, offset);
                    }
                    valChars[i] = (char) b;
                }
                value = new String(valChars);
            }

            // R16a: Name (SAVEPOINT, ROLLBACK)
            int spName = -1;
            if (type == 6 || type == 7) {
                spName = u8(log, payloadOffset + 4);
                if (spName < NAME_BYTE_MIN || spName > NAME_BYTE_MAX) {
                    throw new WalException(Code.BAD_NAME, offset);
                }
            }

            // Transaction checks & actions
            if (type == 1) { // BEGIN
                // R17
                if (seenTxns.contains(txn)) {
                    throw new WalException(Code.DUPLICATE_TXN, offset);
                }
                // R18
                if (openTxns.size() >= MAX_OPEN_TXNS) {
                    throw new WalException(Code.TOO_MANY_OPEN, offset);
                }
                // R20
                seenTxns.add(txn);
                openTxns.put(txn, new TxnState());
            } else {
                // R19
                TxnState ts = openTxns.get(txn);
                if (ts == null) {
                    throw new WalException(Code.TXN_NOT_OPEN, offset);
                }

                switch (type) {
                    case 2 -> { // PUT
                        // R19a
                        if (ts.writes.size() >= MAX_WRITES) {
                            throw new WalException(Code.TXN_TOO_LARGE, offset);
                        }
                        // R21
                        ts.writes.add(new WriteOp(true, key, value));
                    }
                    case 3 -> { // DELETE
                        // R19a
                        if (ts.writes.size() >= MAX_WRITES) {
                            throw new WalException(Code.TXN_TOO_LARGE, offset);
                        }
                        // R21
                        ts.writes.add(new WriteOp(false, key, null));
                    }
                    case 4 -> { // COMMIT
                        // R22
                        for (WriteOp op : ts.writes) {
                            if (op.isPut) {
                                state.put(op.key, op.value);
                            } else {
                                state.remove(op.key);
                            }
                        }
                        committed.add(txn);
                        openTxns.remove(txn);
                    }
                    case 5 -> { // ABORT
                        // R23
                        aborted.add(txn);
                        openTxns.remove(txn);
                    }
                    case 6 -> { // SAVEPOINT
                        // R23a
                        ts.savepoints.put(spName, new Savepoint(ts.writes.size(), offset));
                    }
                    case 7 -> { // ROLLBACK
                        // R19b
                        Savepoint sp = ts.savepoints.get(spName);
                        if (sp == null) {
                            throw new WalException(Code.NO_SAVEPOINT, offset);
                        }
                        // R23b
                        while (ts.writes.size() > sp.position) {
                            ts.writes.remove(ts.writes.size() - 1);
                        }
                        Iterator> it = ts.savepoints.entrySet().iterator();
                        while (it.hasNext()) {
                            Map.Entry<Integer, Savepoint> entry = it.next();
                            if (entry.getKey() != spName && entry.getValue().recordOffset > sp.recordOffset) {
                                it.remove();
                            }
                        }
                    }
                    default -> throw new AssertionError("unreachable");
                }
            }
        }

        List<Long> incomplete = new ArrayList<>(openTxns.keySet());
        Collections.sort(incomplete);

        return new Result(
                Collections.unmodifiableMap(state),
                Collections.unmodifiableList(committed),
                Collections.unmodifiableList(aborted),
                Collections.unmodifiableList(incomplete),
                validLength,
                tornTail
        );
    }

    private static int u8(byte[] data, int offset) {
        return data[offset] & 0xFF;
    }

    private static int u16(byte[] data, int offset) {
        return ((data[offset] & 0xFF) << 8) | (data[offset + 1] & 0xFF);
    }

    private static long u32(byte[] data, int offset) {
        return ((long) (data[offset] & 0xFF) << 24)
                | ((long) (data[offset + 1] & 0xFF) << 16)
                | ((long) (data[offset + 2] & 0xFF) << 8)
                | ((long) (data[offset + 3] & 0xFF));
    }

    private record RecordFrame(int offset, int type, int length) {}

    private record WriteOp(boolean isPut, String key, String value) {}

    private record Savepoint(int position, int recordOffset) {}

    private static final class TxnState {
        final List<WriteOp> writes = new ArrayList<>();
        final Map<Integer, Savepoint> savepoints = new HashMap<>();
    }

    public record Result(Map<String, String> state,
                         List<Long> committed,
                         List<Long> aborted,
                         List<Long> incomplete,
                         int validLength,
                         boolean tornTail) {}

    public enum Code {
        BAD_MAGIC, UNKNOWN_TYPE, RECORD_TOO_LARGE, CHECKSUM_MISMATCH,
        MALFORMED_PAYLOAD, BAD_TXN_ID, BAD_KEY, BAD_VALUE,
        DUPLICATE_TXN, TOO_MANY_OPEN, TXN_NOT_OPEN,
        TXN_TOO_LARGE, BAD_NAME, NO_SAVEPOINT
    }

    public static final class WalException extends Exception {
        private final Code code;
        private final int offset;
        public WalException(Code code, int offset) {
            super(code + "@" + offset);
            this.code = code;
            this.offset = offset;
        }
        public Code code() { return code; }
        public int offset() { return offset; }
    }
}
